% ============================================================
% setup/format.tex —— 正文排版规范（按学位论文要求集中配置）
%
% 本文件由 main.tex 通过 \input 引入（放在 commands 之后）。
% 覆盖内容：
%   1) 章节/节/小节标题：字号、字体、对齐、上下间距、序号与题名间距
%   2) 正文段落：宋体小四、英文 Times New Roman、两端对齐、
%                首行缩进 2 汉字符、固定行距 20 磅、段前段后 0
%   3) 图/表序号与题名：宋体五号、居中、单倍行距、序号与题名空 1 汉字符
%   4) 表达式编号：右对齐、(n) 形式、Times New Roman 五号
%
% 字号对照：三号≈16pt / 四号≈14pt / 小四≈12pt / 五号≈10.5pt
% ============================================================

% ---------- 1. 标题格式（ctex 标题机制）----------
% 说明：正文通过字号钩子把 \baselineskip 固定为 20pt（见下方第 2 节），
% 为避免标题间距受该钩子干扰，这里不再引用 \baselineskip，也不再使用
% \SingleLineSkip 实验性代码，而是直接采用不依赖行距的固定值：
%   章：上下空 40pt（≈2 行）；节：上下空 20pt（≈1 行）；小节同节；三级节 10pt。
% 标题行距由 ctex 依各自字号自然排布（黑体本身即粗，不叠加 \bfseries）。
\ctexset{
   % 各章标题：黑体三号(16pt)居中，上下空 40pt
  % 章序号用中文数字（第一章），序号与题名之间空一个汉字符
  chapter = {
    name      = {第,章},
    number    = \chinese{chapter},   % 若学校要求"第1章"阿拉伯数字，改为 \arabic{chapter}
    format    = \centering\heiti\zihao{3},
    aftername = \quad,               % 序号与题名间空一个汉字符
    beforeskip = {-40pt},            % 上空 40pt（≈2 行）；ctex/section 机制沿用 book 惯例：
                                     % 负值表示"垂直空间"，绝对值即间距（标题另起新页时
                                     % 该空间位于页面顶部）。若写正值会被当作水平缩进，
                                     % 导致标题被推入正文、一页塞入多个章标题（Overfull \vbox）。
    afterskip  = {40pt},             % 下空 40pt（标题后为垂直空间）
    fixskip   = true,                % 固定间距（无弹性）
    % 章首页样式：此处保持 plain（前置部分：页脚居中罗马数字、无页眉）。
    % 正文（\mainmatter 之后）由 main.tex 中的 \ctexset{chapter/pagestyle=fancy}
    % 切换为 fancy，使章首页也有页眉+页码（附件10：正文所有页均有页眉）。
    % 注意：ctex 的 chapter/pagestyle 对 \chapter 与 \chapter* 一视同仁，
    % 若在此处直接写 fancy，前置部分的声明/承诺书/目录等 \chapter* 页面也会被
    % 套上正文页眉（目录第一页会出现"中国地质大学 xvii"），故不能在此写 fancy。
    pagestyle = plain,
  },
  % 一级节标题：黑体四号(14pt)居中，上下空 20pt
  section = {
    format    = \centering\heiti\zihao{4},
    aftername = \quad,
    beforeskip = {20pt},             % 上空 20pt（≈1 行）
    afterskip  = {20pt},             % 下空 20pt
    fixskip   = true,
  },
  % 二级节标题：黑体小四(12pt)居左，上下空 20pt
  subsection = {
    format    = \raggedright\heiti\zihao{-4},
    aftername = \quad,
    beforeskip = {20pt},
    afterskip  = {20pt},
    fixskip   = true,
  },
  % 三级节标题（如使用）：规范表3.8未单独规定，按 CY/T 35-2001 顺延为
  % 黑体小四、顶格（序号与题名间不额外加空格），上下空约半行(10pt)
  subsubsection = {
    format    = \raggedright\heiti\zihao{-4},
    aftername = {},
    beforeskip = {10pt},
    afterskip  = {10pt},
    fixskip   = true,
  },
}

% ---------- 2. 正文段落 ----------
% 宋体小四由文档类 zihao=-4 统一控制；英文 Times New Roman 由 fonts.tex 控制
% 其余：两端对齐、首行缩进 2 汉字符、固定行距 20 磅、段前段后 0
\setlength{\parindent}{2\ccwd}      % 首行左缩进 2 个汉字符
\setlength{\parskip}{0pt}           % 段前段后 0（段与段之间不额外留空）
% 固定值行距 20 磅：直接控制 \baselineskip。
% 方案：用 \g@addto@macro 把 \fixLineskip{20pt} 追加进各字号的内部宏——
%       每次切换字号都会重新执行，行距恒定；\lineskiplimit=-1000pt 为
%       TeX"固定行距"模式，等效 Word"固定值"。
% 注意：\newcommand 名称不能含 @，故公开名 \fixLineskip 不带 @；其定义
%       在 \makeatletter 环境中写入，展开时用到的是普通寄存器与字面量。
\makeatletter
\newcommand{\fixLineskip}[1]{%
  \setlength{\baselineskip}{#1}%
  \setlength{\lineskip}{0pt}%
  \setlength{\lineskiplimit}{-1000\p@}%  % 内核写法：\p@ = pt（此处处于 makeatletter 环境）；负极限使基线间距恒为 baselineskip（固定行距模式）
  \setlength{\normallineskip}{0pt}%
  \setlength{\normallineskiplimit}{-1000\p@}%
  % 注意：不要修改 \topskip！\topskip 是页面首行的全局参数，若在章标题字号
  % 环境中被改成 20pt，ctex 章标题的巨型垂直锚 \topskip0pt 会失效，导致
  % 整页内容堆叠、出现 "Overfull \vbox (474pt too high)"。此处仅同步当前
  % 列表环境的 \linepenalty 相关参数即可。
  \advance\parskip by 0pt%               % 无害的空操作，保持钩子幂等
}
% LaTeX 内核没有可公开使用的 \addto（定义完 \g@addto@macro 后即被释放），
% 必须用 \g@addto@macro 追加钩子。
\g@addto@macro\@normalsize{\fixLineskip{20pt}}
\@ifundefined{@small}{}{\g@addto@macro\@small{\fixLineskip{20pt}}}
\@ifundefined{@footnotesize}{}{\g@addto@macro\@footnotesize{\fixLineskip{20pt}}}
% ctex 的 \zihao{-4} 等字号命令（ctexsize 机制）同样经由 \@setfontsize 定义，
% 因此上面追加进 \@normalsize/\@small/\@footnotesize 的钩子已覆盖全部字号切换；
% 不再重定义 \@setfontsize（旧补丁中的 \xdef\@currsizename{...\protect...}
% 会把 \protect 带进 \csname，导致 "Missing \endcsname inserted" 编译错误）。
\makeatother
\AtBeginDocument{\normalsize}   % 使上述设置立即生效

% ---------- 3. 图 / 表序号与题名 ----------
% 宋体五号(10.5pt)居中，单倍行距，序号与题名之间空一个汉字符
% 图序置于图下方（默认），表序置于表上方
\usepackage{caption}
% 注意：caption 的 font 键不接受 \songti、\zihao{5} 这类"声明式"命令，
% 必须使用 caption 提供的受控字体/字号名（标准尺寸名或 \DeclareCaptionFont
% 自定义注册名）。中文规范"题注宋体五号居中"实现如下：
%   wuhao  = 严格五号（10.5pt/13pt 单倍行距）；
%   zhsongcn = ctex 定义的宋体族控制序列 \songti。caption 的 font 键
%            按名字展开字体命令，因此用 \csname songti\endcsname 间接
%            引用；songti 由 ctex（scheme=chinese）定义。
\DeclareCaptionFont{wuhao}{\fontsize{10.5pt}{13pt}\selectfont}
% 宋体族名：统一使用 ctex 定义的 CJKfamily "zhsong"
% ctex 定义的 CJKfamily "zhsong"（windows/mac/linux/fandol 均注册此名）。
% caption 在正常文档阶段展开，此时 \songti 已由 ctex 定义。
\DeclareCaptionFont{zhsongcn}{\csname songti\endcsname}  % 宋体（ctex 提供）
\captionsetup{
  font         = {zhsongcn,wuhao},      % 宋体五号
  labelfont    = {zhsongcn,wuhao},      % 图/表序号同样宋体五号
  justification = centering,          % 居中
  labelsep     = quad,                % 序号与题名之间空一个汉字符
  skip         = 6pt,                 % 图/表与题名的间距
}
\captionsetup[table]{position=above}   % 表序在表上方
\captionsetup[figure]{position=below}  % 图序在图下方

% ---------- 4. 表达式编号 ----------
% 右对齐、(n) 形式、Times New Roman 五号（默认已带圆括号，仅改字体与字号）
\makeatletter
\def\maketag@@@#1{\hbox{\m@th\normalfont\zihao{5}#1}}
\makeatother

% ---------- 5. 目录排版 ----------
% 标题"目录"：沿用 ctexset 的 chapter 样式（黑体三号加粗居中，单倍行距），无需额外设置。
% 目录条目要求：
%   各章目录  ：宋体四号(14pt)，固定行距 20 磅，两端对齐，页码右对齐，左缩进 0
%   一级节    ：宋体小四(12pt)，固定行距 20 磅，两端对齐，页码右对齐，左缩进 1 汉字符
%   二级节    ：宋体小四(12pt)，固定行距 20 磅，两端对齐，页码右对齐，左缩进 2 汉字符
\usepackage{titletoc}
\titlecontents{chapter}[0pt]{%                        % 左缩进 0
    \zihao{4}\songti\baselineskip=20pt%               % 宋体四号 + 固定 20 磅行距
  }{%
    \thecontentslabel\quad%                           % 编号（第一章）+ 空一个汉字符
  }{%                                                  % 无编号条目（如参考文献/致谢）
    {}%
  }{%
    \hfill\contentspage%                               % 页码右对齐（无点线）
  }
\titlecontents{section}[1\ccwd]{%                      % 左缩进 1 汉字符
    \zihao{-4}\songti\baselineskip=20pt%              % 宋体小四 + 固定 20 磅行距
  }{%
    \thecontentslabel\quad%
  }{%
    {}%
  }{%
    \hfill\contentspage%
  }
\titlecontents{subsection}[2\ccwd]{%                   % 左缩进 2 汉字符
    \zihao{-4}\songti\baselineskip=20pt%
  }{%
    \thecontentslabel\quad%
  }{%
    {}%
  }{%
    \hfill\contentspage%
  }

% 图目录 / 表目录：标题沿用 chapter 样式（黑体三号加粗居中）；
% 条目按一级节样式：宋体小四(12pt)、固定 20 磅、页码右对齐、左缩进 1 汉字符
\titlecontents{figure}[1\ccwd]{%
    \zihao{-4}\songti\baselineskip=20pt%
  }{%
    \thecontentslabel\quad%
  }{}{%
    \hfill\contentspage%
  }
\titlecontents{table}[1\ccwd]{%
    \zihao{-4}\songti\baselineskip=20pt%
  }{%
    \thecontentslabel\quad%
  }{}{%
    \hfill\contentspage%
  }

% ---------- 6. 中英文摘要 ----------
% 标题：中文"摘要"黑体小二(18pt)加粗居中；英文"Abstract" Times New Roman 小二(18pt)加粗居中；单倍行距
% 正文：中文宋体小四 / 英文 TNR 小四，固定 20 磅，段前段后 0（全局已设）
% 关键词："关键词"三字加粗（黑体）/ "Key Words"两词加粗（TNR），字号同正文
\newcommand{\cnabstracttitle}{%
  {\par\vspace{6pt}\centering\baselineskip=24pt\relax\heiti\zihao{-2} 摘要\par}
  \vspace{10pt}\zihao{-4}\songti\noindent\ignorespaces
}
\newcommand{\enabstracttitle}{%
  {\par\vspace{6pt}\centering\baselineskip=24pt\relax\bfseries\zihao{-2} Abstract\par}
  \vspace{10pt}\normalfont\zihao{-4}\noindent\ignorespaces
}

% ---------- 7. 纸张与页眉页脚（中国地质大学硕士论文格式）----------
% 纸张：A4（210×297），白色
% 页边距：上/下 3cm，左/右 3cm，装订线 0cm
% 页眉距顶 2.5cm（headheight=2.5cm，headsep=0.5cm）；页脚距底 2.0cm（footskip=2.0cm）
% 页眉距正文 0.5cm（headsep）；"页眉距顶 2.5cm"由 geometry 的 top=3cm 保证：
% 正文上边距距纸顶 3cm，减去 headsep 0.5cm，页眉基线恰在距顶 2.5cm 处。
% 注意：headheight 是"页眉内容区自身的高度"，不是"距纸顶的距离"！
% 若误写 headheight=2.5cm，每页页眉区会占去 71pt，把版心压缩约 2.5cm，
% 导致骨架文档（标题多、正文少）中多个章标题被挤进同一页并溢出
% （Overfull \vbox 474pt）。标准值取 14pt（可容纳五号中文行）。
\geometry{
  a4paper,
  top        = 3cm,
  bottom     = 3cm,
  left       = 3cm,
  right      = 3cm,
  headheight = 14pt,    % 页眉内容区高度：单行五号汉字足够（正文页眉与英文摘要页眉均为单行）。
                        % 注意：headheight 只决定"页眉框"高度，不改变版心位置（版心由 top/bottom 固定）；
                        % 取 14pt 时页眉基线恰在距纸顶约 2.5cm 处，符合规范"页眉距顶 2.5cm"。
                        % 若日后页眉内容确实需要两行，再增大此值即可（不会挤压版心）。
  headsep    = 0.5cm,   % 页眉底边到正文的距离
  % 注意：geometry 没有 footheight 键（正确键名为 bottomheight / footskip），
  % 原 footheight=1.0cm 属无效选项会被忽略并警告，此处删除；
  % 页脚位置由 footskip 控制。
  footskip   = 2.0cm,   % 正文底边到页脚基线的距离（页脚距底约 2.0cm，贴底）
  bindingoffset = 0cm   % 装订线 0cm
}

% 页眉页脚样式：正文（第一章起）使用 running header
% 按附件10 页眉式样（双面印制）：页码位于【页眉行内】外侧，页脚为空！
%   奇数页：中国地质大学（居中）＋ 页码（行内右侧/外侧）
%   偶数页：页码（行内左侧/外侧）＋ 姓名：题目（居中）
%   页眉之下有下划线（headrule）
% 注意：章首页同样显示页眉（附件10 式样中的页码1即章首页），
%   由上方 \ctexset chapter 的 pagestyle=fancy 实现。
%
% 【重要】必须用 \fancypagestyle{fancy}{...} 封装，不能直接在导言区裸写
% \fancyhead/\fancyhf。原因：fancyhdr 的页眉页脚内容是"全局槽"，
% \fancypagestyle{<样式>} 定义体中的 \fancyhf{} 会在**该样式被 \pagestyle 应用时**
% 清空这些全局槽。英文摘要页用了 \pagestyle{abstracten}（其定义体含 \fancyhf{}），
% 一旦执行就会把导言区设好的正文页眉全部抹掉，导致正文只有页码、页眉空白。
% 封装成 \fancypagestyle{fancy} 后，每次 \pagestyle{fancy} 都会重新装载，不受污染。
\fancypagestyle{fancy}{%
  \fancyhf{}                                   % 清空默认页眉页脚
  \renewcommand{\headrulewidth}{0.4pt}         % 页眉下划线
  \renewcommand{\footrulewidth}{0pt}           % 页脚无线（页码已移入页眉行内）
  \fancyhead[CO]{\songti\zihao{5} 中国地质大学}% 奇数页居中：仅"中国地质大学"（附件10，不含学位类别）
  \fancyhead[RO]{\songti\zihao{5} \thepage}    % 奇数页页码：页眉行内右（外）侧
  \fancyhead[CE]{\songti\zihao{5} \thesisAuthor：\thesisTitle}% 偶数页居中：作者：题目
  \fancyhead[LE]{\songti\zihao{5} \thepage}    % 偶数页页码：页眉行内左（外）侧
  \fancyfoot{}                                 % 页脚为空（附件10 无页脚页码）
}

% 英文摘要页眉：按规范"Abstract 部分用 Times New Roman 五号"，标题居中（TNR）
\fancypagestyle{abstracten}{%
  \fancyhf{}%
  \renewcommand{\headrulewidth}{0.4pt}%
  \fancyhead[C]{\zihao{5} \thesisTitleEN}%      % 英文标题（TNR，主字体即 Times New Roman）
  \fancyfoot[C]{\songti\zihao{5} \thepage}%     % 页码仍按规范用宋体五号
}
